function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet1 = ss.getSheetByName("설문응답") || ss.getSheets()[0];
    var data1 = sheet1.getDataRange().getValues();
    
    var sheet2 = ss.getSheetByName("시트2_특색과목통계");
    var data2 = sheet2 ? sheet2.getDataRange().getValues() : [];
    
    var result = {
      result: "success",
      responses: data1,
      q4Stats: data2
    };
    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet1 = ss.getSheetByName("설문응답");
    if (!sheet1) {
      sheet1 = ss.getSheets()[0];
      sheet1.setName("설문응답");
    }
    
    var data = JSON.parse(e.postData.contents);
    
    if (sheet1.getLastRow() === 0) {
      sheet1.appendRow([
        "제출일시", "자녀 학년", "전반적 만족도", "만족도 이유", 
        "특색교육 만족도", "특색교육 불만족 이유", "희망 특색과목", 
        "감축 희망과목", "증배 희망과목", "희망 방학형태", "재량휴업일 의견", 
        "평가통지 의견", "운동회 만족도", "운동회 건의", "성과발표회 만족도", 
        "성과발표회 건의", "현장체험학습 만족도", "현장체험학습 건의", "종합 건의사항"
      ]);
    }
    
    var timestamp = new Date().toLocaleString("ko-KR", { timeZone: "Asia/Seoul" });
    var gradeStr = data.q1_grade ? data.q1_grade.join(", ") : "";
    var q4Str = data.q4_topics ? data.q4_topics.join(", ") : "";
    
    sheet1.appendRow([
      timestamp,
      gradeStr,
      data.q2_satisfaction || "",
      data.q2_1_reason || "",
      data.q3_special || "",
      data.q3_1_reason || "",
      q4Str,
      data.q5_reduce || "",
      data.q6_increase || "",
      data.q7_vacation || "",
      data.q8_holiday || "",
      data.q9_eval || "",
      data.q10_sports || "",
      data.q10_1_opinion || "",
      data.q11_showcase || "",
      data.q11_1_opinion || "",
      data.q12_trip || "",
      data.q12_1_opinion || "",
      data.q13_overall || ""
    ]);
    
    // 시트 2: Q4 특색과목(활동) 전용 기록
    var sheet2 = ss.getSheetByName("시트2_특색과목통계");
    if (!sheet2) {
      sheet2 = ss.insertSheet("시트2_특색과목통계");
      sheet2.appendRow(["제출일시", "자녀 학년", "선택 특색과목 주제"]);
    }
    
    if (data.q4_topics && data.q4_topics.length > 0) {
      data.q4_topics.forEach(function(topic) {
        sheet2.appendRow([timestamp, gradeStr, topic]);
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({ result: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", error: error.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
